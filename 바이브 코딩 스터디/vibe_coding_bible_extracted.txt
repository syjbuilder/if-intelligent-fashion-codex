const	LIMITS	=	{	free:	10,	pro:	100,	unlimited:	null	};
async	function	getUsageFromToken(accessToken:	string)	{
		//	1.	JWT	→	user	확인
		const	{	user	}	=	await	supabase.auth.getUser(accessToken);
		//	2.	subscriptions	테이블에서	현재	플랜	조회
		const	{	data:	sub	}	=	await	supabase
				.from("subscriptions")

				.eq("user_id",	user.id)
				.maybeSingle();
		const	plan	=	sub?.status	===	"active"	?	sub.plan	:	"free";
		const	limit	=	LIMITS[plan];
		//	3.	이번	달	user	메시지	수	카운트
		const	{	count	}	=	await	supabase
				.from("messages")

				.eq("role",	"user")
				.gte("created_at",	startOfMonthUTC())
				.in("conversation_id",	userConversationIds);
		//	4.	한도	비교
		return	{
				plan,
				used:	count	??	0,
				limit,
				allowed:	limit	===	null	||	count	<	limit
		};
}
